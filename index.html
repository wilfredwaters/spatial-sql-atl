    Session 4 · Who touches whom 
    -- 500 fake phone pings around downtown, one row each
WITH pings AS (
  SELECT i AS ping_id,
         ST_Point(-84.3900 + (hash(i) % 1000) / 1000.0 * 0.04,
                  33.7400  + (hash(i * 7) % 1000) / 1000.0 * 0.03) AS geom
  FROM range(500) t(i)),
zone AS (
  SELECT ST_MakeEnvelope(-84.4000, 33.7450, -84.3800, 33.7650) AS geom)
SELECT p.ping_id, p.geom,
       ST_Within(p.geom, z.geom) AS in_zone
FROM pings p, zone z;

    Roughly a quarter of pings should be in_zone. Widen the zone until half are. Then count them: count(*) FILTER (WHERE in_zone).

    Session 7 · Life along a line 
    -- Snap each fake ping to the nearest point on a route
WITH route AS (
  SELECT ST_MakeLine(ST_Point(-84.3915, 33.7539),
                     ST_Point(-84.3346, 33.9462)) AS line),
pings AS (
  SELECT i AS ping_id,
         ST_Point(-84.3915 + (hash(i) % 1000) / 1000.0 * 0.06,
                  33.7539  + (hash(i * 7) % 1000) / 1000.0 * 0.19) AS geom
  FROM range(40) t(i))
SELECT p.ping_id,
       ST_LineLocatePoint(r.line, p.geom) AS fraction_along,
       ST_LineInterpolatePoint(r.line,
         ST_LineLocatePoint(r.line, p.geom)) AS snapped
FROM pings p, route r;

    fraction_along is 0 to 1. Plot snapped on the map, then ask: what does a ping 2 km from the line become?
